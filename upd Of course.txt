local Players = game:GetService("Players")

local UIModule = {}
UIModule.__index = UIModule

local function new(class, props, children)
	local inst = Instance.new(class)
	for k, v in pairs(props or {}) do inst[k] = v end
	for _, c in ipairs(children or {}) do c.Parent = inst end
	return inst
end

local function corner(r)
	return new("UICorner", { CornerRadius = UDim.new(0, r) })
end

local function softStroke(C)
	return new("UIStroke", { Color = C.accent, Transparency = 0.85, Thickness = 1 })
end

local function tween(obj, t, props, style, dir)
	local tw = TweenService:Create(
		obj,
		TweenInfo.new(t, style or Enum.EasingStyle.Quint, dir or Enum.EasingDirection.Out),
		props
	)
	tw:Play()
	return tw
end

local function loopTween(obj, t, props)
	TweenService:Create(
		obj,
		TweenInfo.new(t, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut, -1, true),
		props
	):Play()
end

local function label(props, C)
	props.BackgroundTransparency = 1
	props.Font = props.Font or Enum.Font.Gotham
	props.TextColor3 = props.TextColor3 or C.text
	props.TextXAlignment = props.TextXAlignment or Enum.TextXAlignment.Left
	return new("TextLabel", props)
end

local function gradientText(lbl, C)
	new("UIGradient", {
		Color = ColorSequence.new(C.accent, C.accent2),
		Parent = lbl,
	})
end

local function hover(btn, normal, over, track)
	track(btn.MouseEnter:Connect(function() tween(btn, 0.15, over) end))
	track(btn.MouseLeave:Connect(function() tween(btn, 0.15, normal) end))
end

local function makeDraggable(handle, target, track)
	local dragging, dragStart, startPos
	track(handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = target.Position
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then dragging = false end
			end)
		end
	end))
	track(UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch) then
			local d = input.Position - dragStart
			target.Position = UDim2.new(
				startPos.X.Scale, startPos.X.Offset + d.X,
				startPos.Y.Scale, startPos.Y.Offset + d.Y
			)
		end
	end))
end

local function copyToClipboard(txt)
	if setclipboard then
		setclipboard(txt)
	elseif toclipboard then
		toclipboard(txt)
	end
end

local function getParent()
	local ok, ui = pcall(function() return gethui and gethui() end)
	if ok and ui then return ui end
	ok, ui = pcall(function() return game:GetService("CoreGui") end)
	if ok and ui then return ui end
	return player:WaitForChild("PlayerGui")
end

function UIModule:SelectTab(t)
	if self.currentTab == t then return end
	for _, o in pairs(self.tabs) do
		local active = (o == t)
		tween(o.btn, 0.2, { BackgroundTransparency = active and 0.1 or 1 })
		tween(o.lbl, 0.2, { TextColor3 = active and Color3.new(1, 1, 1) or self.C.sub })
		if not active then o.page.Visible = false end
	end
	self.currentTab = t
	t.page.Position = UDim2.fromOffset(0, 18)
	t.page.Visible = true
	tween(t.page, 0.35, { Position = UDim2.fromOffset(0, 0) }, Enum.EasingStyle.Back)
end

function UIModule:AddTab(name, icon, order)
	local btn = new("TextButton", {
		Name = name,
		Size = UDim2.new(1, 0, 0, 34),
		BackgroundColor3 = self.C.accent,
		BackgroundTransparency = 1,
		Text = "",
		AutoButtonColor = false,
		LayoutOrder = order or (#self.tabs + 1),
		Parent = self.sideContainer,
	}, { corner(10) })

	local lbl = label({
		Text = (icon or "⭐") .. "  " .. name,
		Font = Enum.Font.GothamBold,
		TextSize = 12,
		TextColor3 = self.C.sub,
		Position = UDim2.fromOffset(10, 0),
		Size = UDim2.new(1, -10, 1, 0),
		Parent = btn,
	}, self.C)

	local page = new("Frame", {
		Name = name .. "Page",
		Size = UDim2.fromScale(1, 1),
		BackgroundTransparency = 1,
		Visible = false,
		ClipsDescendants = false,
		Parent = self.pagesContainer,
	}, {
		new("UIListLayout", { Padding = UDim.new(0, 8), SortOrder = Enum.SortOrder.LayoutOrder }),
	})

	local t = { btn = btn, lbl = lbl, page = page }
	self.tabs[name] = t

	self.track(btn.MouseEnter:Connect(function()
		if self.currentTab ~= t then tween(btn, 0.15, { BackgroundTransparency = 0.8 }) end
	end))
	self.track(btn.MouseLeave:Connect(function()
		if self.currentTab ~= t then tween(btn, 0.15, { BackgroundTransparency = 1 }) end
	end))
	self.track(btn.MouseButton1Click:Connect(function() self:SelectTab(t) end))

	return t
end

function UIModule:AddLinkButton(container, titleText, urlText, btnColor)
	local btn = new("TextButton", {
		Size = UDim2.new(1, 0, 0, 42),
		BackgroundColor3 = self.C.field,
		Text = "",
		AutoButtonColor = false,
		Parent = container,
	}, { corner(8) })

	label({
		Text = titleText,
		Font = Enum.Font.GothamBold,
		TextSize = 12,
		Position = UDim2.fromOffset(12, 4),
		Size = UDim2.new(1, -24, 0, 18),
		Parent = btn,
	}, self.C)

	local urlLbl = label({
		Text = urlText,
		Font = Enum.Font.Gotham,
		TextSize = 10,
		TextColor3 = self.C.sub,
		Position = UDim2.fromOffset(12, 22),
		Size = UDim2.new(1, -24, 0, 14),
		TextTruncate = Enum.TextTruncate.AtEnd,
		Parent = btn,
	}, self.C)

	hover(btn, { BackgroundColor3 = self.C.field }, { BackgroundColor3 = (btnColor or self.C.accent):Lerp(self.C.field, 0.6) }, self.track)

	self.track(btn.MouseButton1Click:Connect(function()
		copyToClipboard(urlText)
		urlLbl.Text = "Copied to Clipboard! ✓"
		urlLbl.TextColor3 = self.C.good
		task.delay(2, function()
			if urlLbl and urlLbl.Parent then
				urlLbl.Text = urlText
				urlLbl.TextColor3 = self.C.sub
			end
		end)
	end))

	return btn
end

function UIModule:destroy()
	self.alive = false
	for _, c in ipairs(self.conns) do
		pcall(function() c:Disconnect() end)
	end
	if self.gui then pcall(function() self.gui:Destroy() end) end
end

return UIModule
